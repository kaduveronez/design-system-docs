## Código Manual

```tsx
import { Pagination, PaginationContent, PaginationItem, PaginationLink } from "@/components/ui/pagination"

export function ExemploPagination() {
  return (
    <Pagination>
      <PaginationContent>
        <PaginationItem><PaginationLink href="#">1</PaginationLink></PaginationItem>
      </PaginationContent>
    </Pagination>
  )
}
```
