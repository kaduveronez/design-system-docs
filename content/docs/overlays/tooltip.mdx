# Tooltip

Rótulo curto que fornece texto explicativo ao passar o cursor do mouse sobre um elemento (Topo, Baixo, Esquerda, Direita).

import {
  Tooltip,
  TooltipContent,
  TooltipTrigger,
} from '@/components/ui/tooltip'
import { Button } from '@/components/ui/button'
import { ComponentPreview } from '@/components/docs/ComponentPreview'

## Código Manual

```tsx
import { Tooltip, TooltipContent, TooltipTrigger } from "@/components/ui/tooltip"
import { Button } from "@/components/ui/button"

export function ExemploTooltip() {
  return (
    <Tooltip>
      <TooltipTrigger render={<Button variant="outline">Topo</Button>} />
      <TooltipContent side="top"><p>Tooltip em Cima</p></TooltipContent>
    </Tooltip>
  )
}
```
