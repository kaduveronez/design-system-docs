# Tooltip

Rótulo curto que fornece texto explicativo ao passar o cursor do mouse sobre um elemento.

import {
  Tooltip,
  TooltipContent,
  TooltipTrigger,
} from '@/components/ui/tooltip'
import { Button } from '@/components/ui/button'
import { Copy } from 'lucide-react'
import { ComponentPreview } from '@/components/docs/ComponentPreview'

## Preview Interativo

Passe o mouse sobre o botão para visualizar a dica flutuante interativa:

## Código Manual

```tsx
import { Tooltip, TooltipContent, TooltipTrigger } from "@/components/ui/tooltip"
import { Button } from "@/components/ui/button"

export function ExemploTooltip() {
  return (
    <Tooltip>
      <TooltipTrigger render={<Button variant="outline">Passe o mouse</Button>} />
      <TooltipContent><p>Dica explicativa</p></TooltipContent>
    </Tooltip>
  )
}
```
