## Código Manual

```tsx
import { Button } from "@/components/ui/button"
import { toast } from "sonner"

export function ToastDemo() {
  return (
    <Button onClick={() => toast("Link copiado com sucesso!")}>
      Copiar Link
    </Button>
  )
}
```
