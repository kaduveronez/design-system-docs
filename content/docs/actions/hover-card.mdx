# Hover Card

Permite visualizar detalhes e ações adicionais quando o usuário passa o mouse sobre um link ou perfil.

## Código Manual

```tsx
import { HoverCard, HoverCardContent, HoverCardTrigger } from "@/components/ui/hover-card"
import { Button } from "@/components/ui/button"

export function ExemploHoverCard() {
  return (
    <HoverCard>
      <HoverCardTrigger render={<Button variant="link">@genqrcode</Button>} />
      <HoverCardContent>
        <p className="text-sm">Gen QRcode Platform</p>
      </HoverCardContent>
    </HoverCard>
  )
}
```
