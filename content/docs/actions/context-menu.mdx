# Context Menu

Exibe um menu de opções em resposta ao clique com o botão direito do mouse em uma área demarcada.

import { ComponentPreview } from '@/components/docs/ComponentPreview'

## Código Manual

```tsx
import {
  ContextMenu,
  ContextMenuContent,
  ContextMenuItem,
  ContextMenuTrigger,
} from "@/components/ui/context-menu"

export function ExemploContextMenu() {
  return (
    <ContextMenu>
      <ContextMenuTrigger className="flex h-[150px] w-[300px] items-center justify-center rounded-md border border-dashed text-sm">
        Clique com o botão direito aqui
      </ContextMenuTrigger>
      <ContextMenuContent>
        <ContextMenuItem>Copiar URL</ContextMenuItem>
        <ContextMenuItem>Baixar QR Code PNG</ContextMenuItem>
      </ContextMenuContent>
    </ContextMenu>
  )
}
```
